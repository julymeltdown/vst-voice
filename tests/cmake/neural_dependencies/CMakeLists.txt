cmake_minimum_required(VERSION 3.20)
project(NeuralDependencyFixture LANGUAGES NONE)
foreach(name seam_neural_synthesis seam_platform seam_core seam_authoring_runtime seam_rendering)
  add_library(${name} INTERFACE)
endforeach()
target_link_libraries(seam_neural_synthesis INTERFACE seam_platform)
target_link_libraries(seam_platform INTERFACE seam_core)
if(SCENARIO STREQUAL "direct")
  target_link_libraries(seam_neural_synthesis INTERFACE seam_authoring_runtime)
elseif(SCENARIO STREQUAL "transitive")
  target_link_libraries(seam_core INTERFACE seam_rendering)
elseif(SCENARIO STREQUAL "link_only")
  target_link_libraries(seam_core INTERFACE "$<LINK_ONLY:seam_authoring_runtime>")
elseif(SCENARIO STREQUAL "alias")
  add_library(hidden ALIAS seam_authoring_runtime)
  target_link_libraries(seam_core INTERFACE hidden)
elseif(SCENARIO STREQUAL "cycle")
  target_link_libraries(seam_core INTERFACE seam_platform)
elseif(NOT SCENARIO STREQUAL "allowed")
  message(FATAL_ERROR "Unknown fixture scenario")
endif()
include("${SEAM_GUARD}")
seam_check_neural_dependency_direction()
